REM ======================================================================
REM  Minecraft Java Edition —— 一键下载 + 启动（纯 cmd 命令脚本）
REM
REM  功能对应要求：
REM    1. 先检查本机是否安装了 Java；没有安装就自动安装 OpenJDK 27
REM       （便携版，解压到游戏目录下的 java 文件夹，不改系统环境变量）
REM    2. 下载优先使用国内高速镜像（BMCLAPI、清华镜像、GitHub 加速站），
REM       镜像拿不到时自动回退 Mojang 官方源
REM    3. 再次启动已经下载过的版本时跳过下载，直接启动游戏
REM    4. 登录账号：默认离线账号（用户名在下面配置区修改），
REM       把 LOGIN 改成 msa 就能用正版微软账号登录
REM    5. 可选安装 Fabric 模组加载器：把 MOD_LOADER 改成 fabric 即可，
REM       装完把 mod 的 jar 放进 MC\game\mods 就能玩（不支持 Forge/NeoForge）
REM
REM  使用方法：
REM    把本 txt 文件复制一份，重命名为  Minecraft.bat  ，双击运行即可；
REM    游戏文件会安装到本文件所在目录下的 MC 文件夹中。
REM    推荐 Windows 10 1803 及以上（自带 curl.exe / tar.exe），
REM    游戏版本建议填 1.13 及以上的正式版，留空则自动用最新正式版。
REM
REM  说明：下载、解压、启动全部用 cmd 的 curl / tar / java 命令完成，
REM        只有解析 Mojang 的 JSON 文件时借用系统自带的 powershell，
REM        不需要安装任何第三方启动器。
REM  提示：游戏窗口旁边那个黑框会滚动 XML 格式日志，这是 Minecraft 自带的
REM        日志格式，属正常现象；启动失败请看 logs\latest.log 里的报错。
REM ======================================================================

REM ============================== 配置区 ==============================
REM  游戏版本：留空 = 自动使用最新正式版；也可以填 1.21.4 或 1.20.1 等

REM  离线登录用户名：3-16 位字母、数字或下划线，不要用中文和空格

REM  Fabric 加载器版本：留空 = 自动选该游戏版本的最新稳定版

REM  登录方式：offline = 离线账号（用上面的 PLAYER_NAME，默认）
REM            msa     = 正版微软账号（首次要在网页上输一次代码，令牌会缓存复用）
set "LOGIN=offline"
REM ====================================================================

echo ======================================================================

echo    游戏目录 : %BASE%

REM ---------------------- 运行环境自检 ----------------------
where curl >nul 2>nul
if errorlevel 1 (
    echo [错误] 没有找到 curl.exe，请升级到 Windows 10 1803 及以上版本。
    goto :FATAL
)
echo %BASE% | findstr /c:"'" >nul
if not errorlevel 1 (
    echo [错误] 文件夹路径里不能有单引号，请把本文件放到别的目录再运行。
    goto :FATAL
)
if not exist "%BASE%" md "%BASE%" >nul 2>nul
pushd "%BASE%"
if errorlevel 1 (
    echo [错误] 无法进入目录 %BASE%
    goto :FATAL
)
for %%d in (%D_TMP% %D_JAVA% %D_VER% %D_LIB% %D_AST% %D_NAT% %D_GAME%) do if not exist "%%d" md "%%d" >nul 2>nul
for %%d in (%D_NAT%\java %D_NAT%\jna %D_NAT%\lwjgl %D_NAT%\netty) do if not exist "%%d" md "%%d" >nul 2>nul

set "CURL_PAR="

REM ============================ 1. Java =============================
call :HEAD "1/7 检查 Java 运行环境"
call :ENSURE_JAVA
if errorlevel 1 goto :FATAL

REM ========================= 2. 版本清单 ============================
call :HEAD "2/7 获取 Minecraft 版本清单"

REM ========================= 3. 解析版本 ============================

if errorlevel 1 goto :FATAL
set "MARK=%D_VER%\!VER!\.ok"
if /i "%MOD_LOADER%"=="fabric" (
    call :HEAD "3.5/7 准备 Fabric 加载器"
    call :SETUP_FABRIC
    if errorlevel 1 goto :FATAL
    set "MARK=%D_VER%\!VER!\.ok-fabric-!FAB_VER!"
)

REM ========================= 4. Java 需求 ===========================
call :HEAD "4/7 检查 Java 版本是否满足该游戏版本"
call :CHECK_JAVA_REQ
if errorlevel 1 goto :FATAL

REM ========================= 4.5 登录账号 ==========================
if /i "%LOGIN%"=="msa" (
    call :HEAD "4.5/7 正版微软账号登录"
    call :MSA_LOGIN
    if errorlevel 1 goto :FATAL

REM ========================= 5/6. 下载文件 ==========================
if exist "!MARK!" (
    echo [跳过] 版本 !VER! 之前已经下载完成，本次直接启动。

    call :HEAD "5/7 下载游戏核心文件（依赖库 + 客户端）"
    call :DOWNLOAD_FILES
    if errorlevel 1 goto :FATAL
    call :EXTRACT_NATIVES
    call :HEAD "6/7 下载游戏资源文件（贴图 + 音效 + 语言）"
    call :DOWNLOAD_ASSETS
    if errorlevel 1 goto :FATAL
    > "!MARK!" echo ok
)

REM ============================ 7. 启动 =============================
call :HEAD "7/7 生成启动参数并启动游戏"
call :MAKE_ARGS
if errorlevel 1 goto :FATAL
echo.
echo    正在启动 Minecraft !VER! ，请稍候……
echo    关闭游戏窗口后本窗口才会结束。
echo.
"%JAVA%" @"%D_TMP%\args.txt"
echo.
echo 游戏已退出。如果启动失败，可以查看日志：logs\latest.log
popd

REM ----------------------------------------------------------------------

:ENSURE_JAVA

    call :JAVA_MAJOR "!JAVA!" JM

for /f "delims=" %%i in ('where java 2^>nul') do if not defined JAVA set "JAVA=%%i"

    if not defined JM set "JAVA="

:JAVA_MAJOR
REM  %1 = java.exe 路径    %2 = 回传主版本号的变量名
set "%~2="
"%~1" -version >"%D_TMP%\javaver.txt" 2>&1
if errorlevel 1 exit /b 0
set "JV="
for /f "tokens=3" %%v in ('findstr /i /c:"version" "%D_TMP%\javaver.txt"') do if not defined JV set "JV=%%~v"
if not defined JV exit /b 0
set "MJ="
for /f "tokens=1 delims=." %%a in ("!JV!") do set "MJ=%%a"

:INSTALL_JDK
REM  %1 = 需要安装的 OpenJDK 主版本号
set "NEEDED=%~1"
echo    正在查询 OpenJDK %NEEDED% 的下载地址 ...
set "PKG="
set "GHLINK="
set "JKDSZ="
del /q "%D_TMP%\jdkurl.txt" >nul 2>nul
%PS% "$ErrorActionPreference='Stop'; $r=Invoke-RestMethod -UseBasicParsing -Uri ('https://api.adoptium.net/v3/assets/latest/%NEEDED%/hotspot?os=windows&architecture=x64&image_type=jdk'); $p=$r[0].binary.package; [IO.File]::WriteAllText('%D_TMP%\jdkurl.txt', ($p.name+'|'+$p.link+'|'+$p.size))" >nul 2>nul
if exist "%D_TMP%\jdkurl.txt" for /f "usebackq tokens=1-3 delims=|" %%a in ("%D_TMP%\jdkurl.txt") do (
    set "PKG=%%a"
    set "GHLINK=%%b"
    set "JKDSZ=%%c"
)
set "U1="
set "U2="
set "U3="
set "U4="
set "U5="
if defined PKG (
    set "U1=https://mirrors.tuna.tsinghua.edu.cn/Adoptium/%NEEDED%/jdk/x64/windows/!PKG!"
    set "U2=https://mirrors.huaweicloud.com/adoptium/%NEEDED%/jdk/x64/windows/!PKG!"
)
if defined GHLINK (
    set "U3=https://ghfast.top/!GHLINK!"
    set "U4=https://gh-proxy.com/!GHLINK!"
    set "U5=https://ghproxy.net/!GHLINK!"
)
set "DL_OUT=%D_TMP%\jdk.zip"
set "DL_EXP=!JKDSZ!"

set "DL_ZIP=1"

if not defined JKDSZ set "DL_MIN=50000000"

    exit /b 1
)
echo    正在解压 OpenJDK ...
if exist "%D_TMP%\jdkx" rd /s /q "%D_TMP%\jdkx" >nul 2>nul

set "JDKDIR="
tar -xf "%D_TMP%\jdk.zip" -C "%D_TMP%\jdkx" >nul 2>nul
for /d %%d in ("%D_TMP%\jdkx\*") do if exist "%%~d\bin\java.exe" if not defined JDKDIR set "JDKDIR=%%~d"
if not defined JDKDIR (
    %PS% "Expand-Archive -Force -Path '%D_TMP%\jdk.zip' -DestinationPath '%D_TMP%\jdkx'" >nul 2>nul
    for /d %%d in ("%D_TMP%\jdkx\*") do if exist "%%~d\bin\java.exe" if not defined JDKDIR set "JDKDIR=%%~d"
)

    echo [错误] OpenJDK 解压后找不到 bin\java.exe，安装失败，请重新运行一次。
    exit /b 1
)
if exist "%D_JAVA%" rd /s /q "%D_JAVA%" >nul 2>nul
move "!JDKDIR!" "%D_JAVA%" >nul 2>nul
if not exist "%D_JAVA%\bin\java.exe" (
    echo [错误] OpenJDK 移动到 %D_JAVA% 目录失败，请重新运行一次。
    exit /b 1
)
"%D_JAVA%\bin\java.exe" -version >nul 2>&1
if errorlevel 1 (
    echo [错误] 安装好的 java.exe 无法运行，请重新运行一次。
    exit /b 1
)
del /q "%D_TMP%\jdk.zip" >nul 2>nul
rd /s /q "%D_TMP%\jdkx" >nul 2>nul
exit /b 0


REM ----------------------------------------------------------------------
REM  获取版本清单、确定要启动的版本

:GET_VERSION
set "VER="
set "MANIFEST=%D_TMP%\version_manifest_v2.json"

    set "DL_OUT=%D_TMP%\version_manifest_v2.json"
    set "DL_EXP=0"
    set "DL_MIN=100000"

    echo [错误] 无法获取 Minecraft 版本清单，请检查网络。
    exit /b 1

    set "VER=%MC_VERSION%"

    %PS% "(Get-Content -Raw '%MANIFEST%' | ConvertFrom-Json).latest.release" >"%D_TMP%\latest.txt" 2>nul
    set /p VER=<"%D_TMP%\latest.txt"
    if "!VER!"=="" (
        echo [错误] 无法解析最新版本号。
        exit /b 1
    )

echo    目标游戏版本 !VER!

set "VJSON=%D_VER%\!VER!\!VER!.json"
if not exist "!VJSON!" (
    echo    下载版本文件 !VER!.json ...
    set "VURL="
    del /q "%D_TMP%\vurl.txt" >nul 2>nul
    %PS% "$m=Get-Content -Raw '%MANIFEST%' | ConvertFrom-Json; $u=($m.versions | Where-Object { $_.id -eq '!VER!' } | Select-Object -First 1); if($u){$u.url}" >"%D_TMP%\vurl.txt" 2>nul
    if exist "%D_TMP%\vurl.txt" set /p VURL=<"%D_TMP%\vurl.txt"
    set "DL_OUT=!VJSON!"
    set "DL_EXP=0"
    set "DL_MIN=1000"
    set "DL_ZIP=0"
    set "DL_URLS=%BMCL%/version/!VER!/json"
    if defined VURL set "DL_URLS=!DL_URLS!;!VURL!"
    call :DL
    if not exist "!VJSON!" (
        echo [错误] 版本 !VER! 不存在或下载失败，请检查版本号是否填写正确。
        exit /b 1
    )
)
exit /b 0


REM ----------------------------------------------------------------------
REM  解析版本文件，生成库文件清单、classpath、Java 需求版本
REM ----------------------------------------------------------------------
:MAKE_PLAN
echo    正在解析库文件与资源索引 ...
%PS% "$j=Get-Content -Raw '%D_VER%\!VER!\!VER!.json' | ConvertFrom-Json; function OK($r){ if(-not $r){return $true}; $res=$false; foreach($x in $r){ $mt=$true; if($x.os){ if($x.os.name -and $x.os.name -ne 'windows'){$mt=$false}; if($x.os.arch -and $x.os.arch -notin @('x64','amd64')){$mt=$false} }; if($x.features){ foreach($p in $x.features.PSObject.Properties){ if($p.Value){$mt=$false} } }; if($mt){ if($x.action -eq 'allow'){$res=$true} else {$res=$false} } }; $res }; function MU($u){ $u -replace '^https://libraries\.minecraft\.net/','https://bmclapi2.bangbang93.com/maven/' }; $cp=@(); $f=@(); $n=@(); foreach($l in $j.libraries){ if(-not (OK $l.rules)){continue}; if($l.name -like '*natives-windows-arm64'){continue}; if($l.name -like '*natives-windows-x86'){continue}; $a=$l.downloads.artifact; if($a){ $p='%D_LIB%\'+$a.path; $cp+=$p; $f+=($p+'|'+(MU $a.url)+'|'+$a.url+'|'+$a.size) }; $c=$l.downloads.classifiers; if($c){ $k=$c.PSObject.Properties | Where-Object { $_.Name -like 'natives-windows*' } | Select-Object -First 1; if($k){ $b=$k.Value; $p='%D_LIB%\'+$b.path; $n+=($p+'|'+(MU $b.url)+'|'+$b.url+'|'+$b.size); $f+=($p+'|'+(MU $b.url)+'|'+$b.url+'|'+$b.size) } } }; if($j.downloads.client){ $p='%D_VER%\'+$j.id+'\'+$j.id+'.jar'; $cp+=$p; $f+=($p+'|https://bmclapi2.bangbang93.com/version/'+$j.id+'/client|'+$j.downloads.client.url+'|'+$j.downloads.client.size) }; if($j.assetIndex){ $p='%D_AST%\indexes\'+$j.assetIndex.id+'.json'; $f+=($p+'|https://bmclapi2.bangbang93.com/v1/packages/'+$j.assetIndex.sha1+'/'+$j.assetIndex.id+'.json|'+$j.assetIndex.url+'|'+$j.assetIndex.size) }; if($j.logging.client){ $lg=$j.logging.client.file; $p='%D_VER%\'+$j.id+'\'+$lg.id; $f+=($p+'|'+$lg.url+'|'+$lg.url+'|'+$lg.size) }; [IO.File]::WriteAllLines('%D_TMP%\files.txt',[string[]]@($f)); [IO.File]::WriteAllLines('%D_TMP%\natives.txt',[string[]]@($n)); [IO.File]::WriteAllText('%D_TMP%\classpath.txt',($cp -join ';')); if($j.javaVersion){ [IO.File]::WriteAllText('%D_TMP%\javamaj.txt',[string]$j.javaVersion.majorVersion) } else { [IO.File]::WriteAllText('%D_TMP%\javamaj.txt','21') }" >nul 2>"%D_TMP%\plan.err"
if not exist "%D_TMP%\files.txt" (
    echo [错误] 解析版本文件失败，文件可能已损坏。
    if exist "%D_TMP%\plan.err" type "%D_TMP%\plan.err"
    exit /b 1
)
exit /b 0


REM ----------------------------------------------------------------------
REM  准备 Fabric 加载器（profile 取自 fabric-meta，库文件走 BMCLAPI 镜像）

if not exist "%D_GAME%\mods" md "%D_GAME%\mods" >nul 2>nul
del /q "%D_TMP%\fabric_ok.txt" >nul 2>nul
echo    正在查询 Fabric 加载器版本 ...
%PS% "$ErrorActionPreference='Stop'; $v='!VER!'; $lv='%FABRIC_LOADER%'; $mk='https://bmclapi2.bangbang93.com/fabric-meta'; $og='https://meta.fabricmc.net'; $cache='%D_TMP%\fabric_profile.json'; $raw=$null; foreach($b in @($mk,$og)){ try { if($lv -eq ''){ $r=Invoke-RestMethod -UseBasicParsing ($b+'/v2/versions/loader/'+$v); $s=@($r | Where-Object { $_.loader.stable }); if($s.Count -gt 0){ $lv=$s[0].loader.version } else { $lv=@($r)[0].loader.version } }; $w=Invoke-WebRequest -UseBasicParsing ($b+'/v2/versions/loader/'+$v+'/'+$lv+'/profile/json'); $raw=$w.Content; break } catch { $lv='%FABRIC_LOADER%' } }; if($raw){ [IO.File]::WriteAllText($cache,$raw) } elseif(Test-Path $cache){ $raw=Get-Content -Raw $cache; Write-Host '    [提示] 连不上 Fabric 服务器，使用本地缓存的加载器配置' } else { Write-Host '[错误] 无法获取 Fabric 加载器配置'; exit 2 }; $p=$raw | ConvertFrom-Json; $dlv=($p.id -replace '^fabric-loader-','') -replace ('-'+[regex]::Escape($v)+'$'),''; if($dlv){ $lv=$dlv }; $fl=@(); $fl+=('LOADER='+$lv); $fl+=('MAINCLASS='+$p.mainClass); foreach($a in @($p.arguments.jvm)){ if($a){ $fl+=('JVMA='+$a) } }; $files=@(); $libs=@(); $ids=@(); foreach($l in @($p.libraries)){ $n=[string]$l.name; $q=$n -split ':'; if($q.Count -lt 3){ continue }; $rel=($q[0] -replace '\.','/')+'/'+$q[1]+'/'+$q[2]+'/'+$q[1]+'-'+$q[2]; if($q.Count -gt 3){ $rel=$rel+'-'+$q[3] }; $rel=$rel+'.jar'; $dst='%D_LIB%\'+($rel -replace '/','\'); $libs+=$dst; $sg=$rel -split '/'; $ids+=(($sg[0..($sg.Count-3)]) -join '/'); $repo='https://maven.fabricmc.net/'; if($l.url){ $repo=[string]$l.url }; if(-not $repo.EndsWith('/')){ $repo=$repo+'/' }; $files+=($dst+'|https://bmclapi2.bangbang93.com/maven/'+$rel+'|'+$repo+$rel); $fl+=('LIB='+$n) }; [IO.File]::WriteAllText('%D_TMP%\fabric.txt', ($fl -join [char]10)); [IO.File]::WriteAllLines('%D_TMP%\fabric_files.txt', [string[]]@($files)); $cp=(Get-Content -Raw '%D_TMP%\classpath.txt').Trim(); $parts=@($cp -split ';'); $client=$parts[-1]; $rest=@(); if($parts.Count -gt 1){ $rest=$parts[0..($parts.Count-2)] }; $keep=@(); foreach($c in $rest){ $r2=($c -replace '^%D_LIB%\\','' -replace '\\','/'); $dup=$false; foreach($id in $ids){ if($r2.StartsWith($id+'/')){ $dup=$true; break } }; if(-not $dup){ $keep+=$c } }; $rest=$keep; [IO.File]::WriteAllText('%D_TMP%\classpath.txt', ((@($rest)+@($libs)+@($client)) -join ';')); [IO.File]::WriteAllText('%D_TMP%\fabric_ok.txt', $lv); Write-Host ('    已准备 Fabric ' + $lv + '，加载器库 ' + $files.Count + ' 个')"
set "FAB_VER="
if exist "%D_TMP%\fabric_ok.txt" set /p FAB_VER=<"%D_TMP%\fabric_ok.txt"
if "!FAB_VER!"=="" (
